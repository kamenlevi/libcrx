cmake_minimum_required(VERSION 3.16)
project(libcrx C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
option(CRX_SANITIZE "Build with address and undefined behaviour sanitizers" OFF)
option(CRX_WERROR "Warnings are errors" ON)
set(CRX_LIBRAW_DIR "" CACHE PATH "Prefix with include/libraw and lib/libraw_r.a, for crxoracle")

add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion)
if(CRX_WERROR)
  add_compile_options(-Werror)
endif()
if(CRX_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g -O1)
  add_link_options(-fsanitize=address,undefined)
endif()

add_library(crx STATIC src/crx.c src/container.c src/headers.c src/rice.c src/lines.c src/decode.c src/wavelet.c src/qp.c src/pool.c)
find_package(Threads REQUIRED)
target_link_libraries(crx PUBLIC Threads::Threads)
target_compile_options(crx PRIVATE -fwrapv)
target_include_directories(crx PUBLIC include)

add_library(crxtools STATIC tools/sha256.c)
target_include_directories(crxtools PUBLIC tools)

add_executable(crxcheck tools/crxcheck.c)
target_link_libraries(crxcheck crx crxtools)
add_executable(crxbench tools/crxbench.c)
target_link_libraries(crxbench crx crxtools)

# crxoracle needs LibRaw: a vendored prefix, or pkg-config on Linux.
set(HAVE_LIBRAW OFF)
if(CRX_LIBRAW_DIR)
  add_executable(crxoracle tools/crxoracle.c)
  target_include_directories(crxoracle PRIVATE ${CRX_LIBRAW_DIR}/include)
  target_link_libraries(crxoracle crxtools ${CRX_LIBRAW_DIR}/lib/libraw_r.a)
  if(APPLE)
    target_link_libraries(crxoracle ${CRX_LIBRAW_DIR}/lib/libomp.a ${CRX_LIBRAW_DIR}/lib/libjpeg.a z c++)
  else()
    target_link_libraries(crxoracle gomp jpeg z stdc++ m)
  endif()
  set(HAVE_LIBRAW ON)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(LIBRAW QUIET libraw_r)
    if(LIBRAW_FOUND)
      add_executable(crxoracle tools/crxoracle.c)
      target_include_directories(crxoracle PRIVATE ${LIBRAW_INCLUDE_DIRS})
      target_link_libraries(crxoracle crxtools ${LIBRAW_LIBRARIES})
      target_link_directories(crxoracle PRIVATE ${LIBRAW_LIBRARY_DIRS})
      set(HAVE_LIBRAW ON)
    endif()
  endif()
endif()
message(STATUS "crxoracle (needs LibRaw): ${HAVE_LIBRAW}")

add_executable(mutate fuzz/mutate.c)
target_link_libraries(mutate crx)
